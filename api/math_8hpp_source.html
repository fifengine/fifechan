403 template <typename T>
404 inline T Math<T>::ACos(T _val)
405 {
406 if (-static_cast<T>(1) < _val) {
407 if (_val < static_cast<T>(1)) {
408 return static_cast<T>(std::acos(_val));
409 }
410
411 return static_cast<T>(0);
412 }
413
414 return pi();
415 }
